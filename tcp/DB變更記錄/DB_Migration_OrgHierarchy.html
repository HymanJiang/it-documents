TCP Sales Org Hierarchy

業務組織階層資料表建立與欄位調整
執行對象MFG25、MFG26 …（年份別 DB）
類型建表 + 欄位異動
來源03_DB_Migration / DB_Migration_OrgHierarchy.sql
Step 1　建立資料表（若尚未存在）
建立 TCP_Sales_Org_Node，含自參照的 Parent_Id 階層外鍵與 Email/Site、Parent_Id 索引。
-- Step 1：建立資料表（若尚未存在）
IF NOT EXISTS (SELECT 1 FROM sys.tables WHERE name = 'TCP_Sales_Org_Node')
BEGIN
    CREATE TABLE TCP_Sales_Org_Node (
        Id           UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID(),
        Site         NVARCHAR(50)     NOT NULL,
        Email        NVARCHAR(200)    NOT NULL,
        Display_Name NVARCHAR(200)    NULL,
        RoleIndex    UNIQUEIDENTIFIER NULL,   -- 對應 [ACLDW].[dbo].[RolePermission_Role].RoleIndex（邏輯參照，跨庫不建 FK）
        Parent_Id    UNIQUEIDENTIFIER NULL,
        Sort_Order   INT              NULL,
        CreateDate   DATETIME         NOT NULL DEFAULT GETDATE(),
        CreateUser   NVARCHAR(200)    NOT NULL,
        UpdateDate   DATETIME         NULL,
        UpdateUser   NVARCHAR(200)    NULL,

        CONSTRAINT PK_TCP_Sales_Org_Node PRIMARY KEY (Id),
        CONSTRAINT FK_TCP_Sales_Org_Node_Parent FOREIGN KEY (Parent_Id)
            REFERENCES TCP_Sales_Org_Node (Id)
    );

    CREATE INDEX IX_TCP_Sales_Org_Node_Email_Site ON TCP_Sales_Org_Node (Email, Site);
    CREATE INDEX IX_TCP_Sales_Org_Node_Parent_Id  ON TCP_Sales_Org_Node (Parent_Id);
END
GO
Step 2　移除 IsActive 欄位（若存在舊版欄位）
IsActive 帶有 Default Constraint，需先動態查出並移除 Constraint，才能 DROP COLUMN。
-- Step 2：移除 IsActive 欄位（若存在舊版欄位）
-- IsActive 有 Default Constraint，需先移除 Constraint 才能 DROP COLUMN
IF EXISTS (
    SELECT 1 FROM sys.columns
    WHERE  object_id = OBJECT_ID('TCP_Sales_Org_Node') AND name = 'IsActive'
)
BEGIN
    DECLARE @constraintName NVARCHAR(200);

    SELECT @constraintName = dc.name
    FROM   sys.default_constraints dc
    JOIN   sys.columns c
           ON dc.parent_object_id = c.object_id AND dc.parent_column_id = c.column_id
    WHERE  c.object_id = OBJECT_ID('TCP_Sales_Org_Node') AND c.name = 'IsActive';

    IF @constraintName IS NOT NULL
        EXEC('ALTER TABLE TCP_Sales_Org_Node DROP CONSTRAINT ' + @constraintName);

    ALTER TABLE TCP_Sales_Org_Node DROP COLUMN IsActive;
END
GO
Step 3　Role_Name (NVARCHAR) → RoleIndex (UNIQUEIDENTIFIER)
角色改為對應 [ACLDW].[dbo].[RolePermission_Role].RoleIndex。表內目前為草稿／空資料，直接換欄位、不做資料搬移。
-- Step 3：Role_Name(NVARCHAR) → RoleIndex(UNIQUEIDENTIFIER)
-- 角色改為對應 [ACLDW].[dbo].[RolePermission_Role].RoleIndex。
-- 注意：表內目前為草稿/空資料，直接換欄位、不做資料搬移。
IF EXISTS (
    SELECT 1 FROM sys.columns
    WHERE  object_id = OBJECT_ID('TCP_Sales_Org_Node') AND name = 'Role_Name'
)
BEGIN
    ALTER TABLE TCP_Sales_Org_Node DROP COLUMN Role_Name;
END
GO

IF NOT EXISTS (
    SELECT 1 FROM sys.columns
    WHERE  object_id = OBJECT_ID('TCP_Sales_Org_Node') AND name = 'RoleIndex'
)
BEGIN
    ALTER TABLE TCP_Sales_Org_Node ADD RoleIndex UNIQUEIDENTIFIER NULL;
END
GO
