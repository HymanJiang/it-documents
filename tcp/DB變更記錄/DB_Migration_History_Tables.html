TCP 改版 / DB 變更記錄

上傳歷史快照表

每次 Import 後將本次資料寫入 History 表（只增不刪）
執行對象MFG2025、MFG2026（及各年份 DB）
日期2026-06-05
類型建表（4 張 History 表）
來源03_DB_Migration / DB_Migration_History_Tables.sql
四張 History 表均以 LogId 外鍵關聯 TCP_Upload_Log(Id)，並建立 LogId 索引。每張表皆以 IF NOT EXISTS 包覆，可重複執行。
1. TCP_Sales_TCR_Rate_History
CREATE TABLE TCP_Sales_TCR_Rate_History (
    Id UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    LogId UNIQUEIDENTIFIER NOT NULL,
    Site NVARCHAR(100) NULL, Email NVARCHAR(150) NULL,
    Group_Name NVARCHAR(50) NULL, Commission_Type NVARCHAR(20) NULL,
    Ach_Threshold DECIMAL(18,4) NULL, Ach_Threshold_Monthly DECIMAL(18,4) NULL,
    GP DECIMAL(8,4) NULL,
    Standard_CR DECIMAL(18,4) NULL,   -- 及 Standard_CR_Jan ~ Dec
    Monthly_CR  DECIMAL(18,4) NULL,   -- 及 Monthly_CR_Jan ~ Dec
    CreateDate DATETIME NOT NULL DEFAULT GETDATE(),
    CreateUser NVARCHAR(100) NOT NULL,
    CONSTRAINT FK_Rate_History_Log FOREIGN KEY (LogId) REFERENCES TCP_Upload_Log(Id)
);
CREATE INDEX IX_Rate_History_LogId ON TCP_Sales_TCR_Rate_History(LogId);
2. TCP_Sales_TCR_SalesIDConfig_History
CREATE TABLE TCP_Sales_TCR_SalesIDConfig_History (
    Id UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    LogId UNIQUEIDENTIFIER NOT NULL,
    Email NVARCHAR(150) NULL, Group_Name NVARCHAR(50) NULL, Sales_Id NVARCHAR(50) NULL,
    Jan INT NULL, ... [Dec] INT NULL,   -- 各月份啟用旗標
    CreateDate DATETIME NOT NULL DEFAULT GETDATE(), CreateUser NVARCHAR(100) NOT NULL,
    CONSTRAINT FK_SIDConfig_History_Log FOREIGN KEY (LogId) REFERENCES TCP_Upload_Log(Id)
);
3. TCP_Sales_TCR_Mapping_History
CREATE TABLE TCP_Sales_TCR_Mapping_History (
    Id UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    LogId UNIQUEIDENTIFIER NOT NULL,
    Email NVARCHAR(150) NULL, Sales_Email NVARCHAR(150) NULL,
    CreateDate DATETIME NOT NULL DEFAULT GETDATE(), CreateUser NVARCHAR(100) NOT NULL,
    CONSTRAINT FK_Mapping_History_Log FOREIGN KEY (LogId) REFERENCES TCP_Upload_Log(Id)
);
4. TCP_Sales_RS_History
CREATE TABLE TCP_Sales_RS_History (
    Id UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID() PRIMARY KEY,
    LogId UNIQUEIDENTIFIER NOT NULL,
    Email NVARCHAR(150) NULL, Group_Name NVARCHAR(50) NULL, Report NVARCHAR(50) NULL,
    FOB_Local NVARCHAR(20) NULL, Region NVARCHAR(50) NULL, Entity NVARCHAR(100) NULL,
    Sector NVARCHAR(100) NULL, Prod_Group NVARCHAR(100) NULL,
    Jan INT NULL, ... [Dec] INT NULL,
    CreateDate DATETIME NOT NULL DEFAULT GETDATE(), CreateUser NVARCHAR(100) NOT NULL,
    CONSTRAINT FK_RS_History_Log FOREIGN KEY (LogId) REFERENCES TCP_Upload_Log(Id)
);
