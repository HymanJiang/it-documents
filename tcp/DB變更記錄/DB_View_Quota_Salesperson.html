Quota Views（業務員）

有效 Sales ID 計算 + 結合幣別與 Quota 的業務員別 View
目標 DBaclemanager-db.MFG26（及各年份 MFG DB）
日期2026-06-05
類型建立 2 個 View
來源03_DB_Migration / DB_View_Quota_Salesperson.sql
View 1：vw_TCP_Effective_SalesId
依 Commission Type 計算每位業務員的有效 Sales ID 清單：IND/NOACC = idmap 基底 + SalesIDConfig Delta；LEADER = Org Hierarchy（遞迴）+ idmap 基底 + SalesIDConfig Delta。SalesIDConfig 全零視為排除、至少一月有值視為新增。
CREATE VIEW vw_TCP_Effective_SalesId AS
WITH
OrgHierarchy AS (   -- 遞迴取得每位 Leader 底下所有子孫節點 Email
    SELECT Id, Email, Site, Parent_Id, Email AS LeaderEmail
    FROM   TCP_Sales_Org_Node
    UNION ALL
    SELECT n.Id, n.Email, n.Site, n.Parent_Id, h.LeaderEmail
    FROM   TCP_Sales_Org_Node n
    JOIN   OrgHierarchy h ON n.Parent_Id = h.Id AND n.Site = h.Site
),
LeaderBase AS (     -- LEADER 的 Org + idmap 基底（未套 Delta）
    SELECT DISTINCT r.Email AS LeaderEmail, ISNULL(r.Group_Name,'') AS Group_Name,
           r.Site, i.Id_SAP AS SalesId
    FROM   TCP_Sales_TCR_Rate r
    JOIN   OrgHierarchy h ON h.LeaderEmail = r.Email AND h.Site = r.Site
    JOIN   idmap i ON i.id_email = h.Email
    WHERE  r.Commission_Type = 'LEADER'
)
-- IND/NOACC 基底（排除 SalesIDConfig 全零）
SELECT r.Email, ISNULL(r.Group_Name,'') AS Group_Name, r.Commission_Type, r.Site, i.Id_SAP AS SalesId
FROM TCP_Sales_TCR_Rate r JOIN idmap i ON i.id_email = r.Email
WHERE r.Commission_Type IN ('IND','NOACC')
  AND NOT EXISTS ( /* SalesIDConfig 12 月加總 = 0 則排除 */ )
UNION ALL
-- IND/NOACC SalesIDConfig 新增（不在 idmap、至少一月有值）
SELECT ... FROM TCP_Sales_TCR_Rate r JOIN TCP_Sales_TCR_SalesIDConfig s ...
UNION ALL
-- LEADER 基底（排除全零）
SELECT lb.LeaderEmail AS Email, lb.Group_Name, 'LEADER', lb.Site, lb.SalesId FROM LeaderBase lb ...
UNION ALL
-- LEADER SalesIDConfig 新增
SELECT ... ;
GO
View 2：vw_TCP_Quota_By_Salesperson
結合有效 Sales ID、幣別邏輯與 Quota 資料。幣別優先序：ReportCurrency → TCP_Site.LocalCurrency。IND/NOACC/LEADER 走 Quota_Sales_Force；RS 類型依報表維度（SalesByRegion / SalesByProduct / MySalesForce）分別 JOIN 對應 Quota 表。
CREATE VIEW vw_TCP_Quota_By_Salesperson AS
-- IND / NOACC / LEADER：從有效 Sales ID 查 Quota
SELECT e.Email, e.Group_Name, e.Commission_Type, e.Site,
       COALESCE(r.ReportCurrency, t.LocalCurrency) AS Currency,
       q.Year_id AS YearId, SUM(q.Jan) AS Jan, ... SUM(q.Total) AS Total
FROM vw_TCP_Effective_SalesId e
JOIN TCP_Sales_TCR_Rate r ON r.Email=e.Email AND ISNULL(r.Group_Name,'')=e.Group_Name
                         AND r.Site=e.Site AND r.Commission_Type=e.Commission_Type
JOIN TCP_Site t ON t.SiteCode = e.Site
JOIN [ACLSTNR11...].[MFG26].[dbo].[Quota_Sales_Force] q
     ON q.Sales_id=e.SalesId AND q.Tran_type='General'
     AND q.Currency = COALESCE(r.ReportCurrency, t.LocalCurrency)
GROUP BY e.Email, e.Group_Name, e.Commission_Type, e.Site,
         COALESCE(r.ReportCurrency,t.LocalCurrency), q.Year_id
UNION ALL
-- RS：SalesByRegion → Sales_Sector
SELECT ... FROM TCP_Sales_TCR_Rate r JOIN TCP_Site t JOIN TCP_Sales_RS rs
       JOIN Sales_Sector q ON ... WHERE r.Commission_Type='RS' AND rs.Report='SalesByRegion' ...
UNION ALL
-- RS：SalesByProduct → SALES_PRODUCT
SELECT ... WHERE rs.Report='SalesByProduct' ...
UNION ALL
-- RS：MySalesForce → SALES_FORCE
SELECT ... WHERE rs.Report='MySalesForce' ;
GO
